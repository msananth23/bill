from kivymd.uix.screen import MDScreen
from kivy.lang import Builder
from kivy.app import App
from kivy.properties import StringProperty
import os

Builder.load_file(os.path.join(os.path.dirname(__file__), "dashboard.kv"))


class DashboardScreen(MDScreen):
    SCREEN_NAME = "dashboard"
    sync_status_text = StringProperty("Checking connection...")

    def on_enter(self, *args):
        self.refresh_status()

    def refresh_status(self):
        app = App.get_running_app()
        if not app.local_db.get_setting("server_url"):
            self.sync_status_text = "No server configured (offline mode) - set one in Settings"
            return
        self.sync_status_text = "Syncing..."
        app.run_sync(on_done=self._on_sync_done)

    def _on_sync_done(self, summary):
        if summary.get("online"):
            renum = summary.get("renumbered", 0)
            msg = f"Online - synced ({summary.get('confirmed', 0)} confirmed"
            if renum:
                msg += f", {renum} renumbered - check Documents)"
            else:
                msg += ")"
            self.sync_status_text = msg
        else:
            self.sync_status_text = "Offline - working from local data, will sync automatically later"

    def new_document(self, doc_type):
        App.get_running_app().go("document_editor", doc_type=doc_type)

    def open_profiles(self):
        App.get_running_app().go("profiles")

    def open_settings(self):
        App.get_running_app().go("settings")
